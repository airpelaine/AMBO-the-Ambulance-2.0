/** * Handles background data transmission (POST requests) initiated from the game's validation engine. */ function doPost(e) { try { var data = JSON.parse(e.postData.contents); logToSheet(data); return ContentService.createTextOutput("Success POST"); } catch(err) { return ContentService.createTextOutput("Error POST: " + err.message); } } /** * Handles front-end redirect actions (GET requests) as a structural fallback. */ function doGet(e) { try { var data = { code: e.parameter.code, id: e.parameter.id, stab: e.parameter.stab, sync: e.parameter.sync, total: e.parameter.total, mod: e.parameter.mod, logs: e.parameter.logs ? JSON.parse(decodeURIComponent(e.parameter.logs)) : [] }; logToSheet(data); return ContentService.createTextOutput("Data Handshake Validated Natively"); } catch(err) { return ContentService.createTextOutput("Error Processing Sync: " + err.message); } } /** * Internal logic engine to map incoming payloads directly to custom module sheets * and sequentially print individual question metrics including the verification code. */ function logToSheet(data) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // Dynamic target mapping structure matrix to translate game module dropdown values var sheetMap = { "acute_care": "ACUTE CARE", "emergency": "EMERGENCY CARE", "ambulatory": "AMBULATORY CARE", "basic": "BASIC NAVIGATION", "acute": "ACUTE SURGICAL CARE", "shiftmanager": "SHIFT MANAGER ROLES", "discharge": "DISCHARGE OFFICER" }; var targetSheetName = sheetMap[data.mod] || "GENERAL LOGS"; var sheet = ss.getSheetByName(targetSheetName); // Fail-safe: Auto-generates the sheet tab with structural headers if it doesn't exist yet if (!sheet) { sheet = ss.insertSheet(targetSheetName); sheet.appendRow(["Date / Time", "Employee #", "Unique Verification Code", "Q1"]); } // Format current St. Luke's standardized timestamp string (e.g., 05/18/26 / 0915H) var timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "MM/dd/yy / HHmm'H'"); // Initialize the baseline row payload array including the unique verification code var newRow = [timestamp, data.id, data.code]; // Unpack and loop through question logs array consecutively (Initial attempt results only) if (data.logs && Array.isArray(data.logs)) { for (var i = 0; i < data.logs.length; i++) { newRow.push(data.logs[i]); } } // Commit the complete record vector to the target module sheet sheet.appendRow(newRow); }
